slot ShippingCalculator {
  @external
  Quote(address: Address, subtotal: Int) -> ShippingQuote throws ShippingUnavailable
}

action CreateShipment(subtotal: Int) -> Shipment {
  let quote = slot ShippingCalculator.Quote(subtotal) else ShippingUnavailable
  return quote
}

scenario "Stub slot" {
  given slot ShippingCalculator.Quote(100) returns ShippingQuote {
    fee: 15
  }
  given slot ShippingCalculator.Quote throws ShippingUnavailable
  expect error ShippingUnavailable
}
