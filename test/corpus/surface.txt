  given Product {
    id: "1",
    stock: 5
  }
  given slot PaymentGateway.Charge returns "ch_1"
  when Run(Input {
    request_id: "r"

  expect result == true
  expect stored Product where id == "1" {
    stock: 3
  }
  expect not_stored Product where id == "2"
  expect emitted OrderPlaced {
    order_id: "o"

  expect error InsufficientStock
