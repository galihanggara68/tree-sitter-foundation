entity Wallet {
  id: uuid @pk
  balance: decimal(12, 2)
}

error WalletNotFound {
  kind: not_found
  message: "Wallet does not exist"
}

action Transfer(from_id: uuid, input: TransferInput) -> bool {
  let recipient = query one Wallet where id == input.to_wallet_id else WalletNotFound
  guard recipient.balance >= input.amount else InsufficientFunds
  update Wallet where id == from_id set {
    balance: recipient.balance - input.amount
  }
  return true
}
